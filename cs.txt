int[][] arr = new int[3][];
for (int i = 0; i < 3; i++)
{
    arr[i] = new int[3];
}

// Read inputs from the console
for (int i = 0; i < 3; i++)
{
    for (int j = 0; j < 3; j++)
    {
        arr[i][j] = Convert.ToInt32(Console.ReadLine());
    }
}

// Print the array matrix
for (int i = 0; i < 3; i++)
{
    for (int j = 0; j < 3; j++)
    {
        Console.Write(arr[i][j] + " ");
    }
    Console.WriteLine();
}

Console.ReadKey();
